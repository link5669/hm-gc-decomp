cmake_minimum_required(VERSION 3.16)
project(hauntedmansion_native C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(HM_GEN_DIR "" CACHE PATH "recompiled C (from recomp/ppc2c.py)")
set(HM_DOL "" CACHE FILEPATH "main_mp.dol to embed")
option(HM_WITH_GPU "link the GX/OpenGL module" ON)
option(HM_WITH_AUDIO "link the audio module" ON)
set(SDL2_ROOT /home/mi/Decomp/ref/sdl2 CACHE PATH "static SDL2 install")
add_library(sdl2static INTERFACE)
target_include_directories(sdl2static INTERFACE ${SDL2_ROOT}/include/SDL2)
target_link_libraries(sdl2static INTERFACE ${SDL2_ROOT}/lib/libSDL2.a /usr/lib/x86_64-linux-gnu/libasound.so.2 pthread dl m)

file(GLOB GEN_SRC ${HM_GEN_DIR}/gen_*.c)
add_library(hmgen STATIC ${GEN_SRC} ${HM_GEN_DIR}/table.c)
target_include_directories(hmgen PRIVATE ${HM_GEN_DIR} runtime)
# guest integer arithmetic wraps; no FP contraction beyond what the guest code asked for
target_compile_options(hmgen PRIVATE -O2 -fwrapv -ffp-contract=off -fno-math-errno -w)

configure_file(runtime/dol.S.in ${CMAKE_BINARY_DIR}/dol.S @ONLY)
set(RT_SRC runtime/main.cpp runtime/cpu.cpp runtime/os.cpp runtime/hw.cpp runtime/events.cpp
           runtime/boot.cpp runtime/hle.cpp runtime/card.cpp ${CMAKE_BINARY_DIR}/dol.S
           net/proto.cpp net/host.cpp net/client.cpp net/stb_impl.cpp)   # net/: online co-op
set(HM_LIBS)
if(HM_WITH_GPU AND EXISTS ${CMAKE_SOURCE_DIR}/gpu/CMakeLists.txt)
  add_subdirectory(gpu)
  list(APPEND HM_LIBS hmgpu)
else()
  list(APPEND RT_SRC runtime/stubs_gpu.cpp)
endif()
if(HM_WITH_AUDIO AND EXISTS ${CMAKE_SOURCE_DIR}/audio/CMakeLists.txt)
  add_subdirectory(audio)
  list(APPEND HM_LIBS hmaudio)
else()
  list(APPEND RT_SRC runtime/stubs_audio.cpp)
endif()
add_executable(hauntedmansion ${RT_SRC})
target_include_directories(hauntedmansion PRIVATE runtime)
target_compile_options(hauntedmansion PRIVATE -O2 -fwrapv -Wall -Wno-unused-function -Wno-misleading-indentation)
target_link_libraries(hauntedmansion PRIVATE hmgen ${HM_LIBS} sdl2static pthread dl m -static-libstdc++ -static-libgcc)
